def Enqueue(Queue, Head, Tail, NumItems, InputData): 
    if NumItems >= 10: 
        return (False, Queue, Head, Tail, NumItems) 
    Queue[Tail] = InputData 
    if Tail >= 9: 
        Tail = 0 
    else: 
        Tail = Tail + 1 
    NumItems = NumItems + 1 
    return (True, Queue, Head, Tail, NumItems)
